\section{USAMO 1975}
        \begin{enumerate}
        	\item (\textit{USAMO 1975}) (a) Prove that


 $[5x]+[5y]\ge [3x+y]+[3y+x]$,

 where $x,y\ge 0$ and $[u]$ denotes the greatest integer $\le u$ (e.g., $[\sqrt{2}]=1$).


 (b) Using (a) or otherwise, prove that


 $\frac{(5m)!(5n)!}{m!n!(3m+n)!(3n+m)!}$

 is integral for all positive integral $m$ and $n$.


 

	\item (\textit{USAMO 1975}) Let $A,B,C,D$ denote four points in space and $AB$ the distance between $A$ and $B$, and so on. Show that

 \[AC^2+BD^2+AD^2+BC^2\ge AB^2+CD^2.\]


 

	\item (\textit{USAMO 1975}) If $P(x)$ denotes a polynomial of degree $n$ such that $P(k)=k/(k+1)$ for $k=0,1,2,\ldots,n$, determine $P(n+1)$.


 

	\item (\textit{USAMO 1975}) Two given circles intersect in two points $P$ and $Q$. Show how to construct a segment $AB$ passing through $P$ and terminating on the two circles such that $AP\cdot PB$ is a maximum.

	\item (\textit{USAMO 1975}) A deck of $n$ playing cards, which contains three aces, is shuffled at random (it is assumed that all possible card distributions are equally likely). The cards are then turned up one by one from the top until the second ace appears. Prove that the expected (average) number of cards to be turned up is $(n+1)/2$.


 

\end{enumerate}
