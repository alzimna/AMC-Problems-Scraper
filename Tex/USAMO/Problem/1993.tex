\section{USAMO 1993}
        \begin{enumerate}
        	\item (\textit{USAMO 1993}) For each integer $n\ge2$, determine, with proof, which of the two positive real numbers $a$ and $b$ satisfying


 $a^n = a + 1, \quad b^{2n} = b + 3a$

 is larger.


 

	\item (\textit{USAMO 1993}) Let $ABCD$ be a convex quadrilateral such that diagonals $AC$ and $BD$ intersect at right angles, and let $E$ be their intersection. Prove that the reflections of $E$ across $AB$, $BC$, $CD$, $DA$ are concyclic.


 

	\item (\textit{USAMO 1993}) Consider functions $f : [0, 1] \rightarrow \mathbb{R}$ which satisfy


 <table><tbody><tr>
<td>     </td><td>(i)</td><td>$f(x)\ge0$ for all $x$ in $[0, 1]$,</td></tr>
<tr>
<td>     </td><td>(ii)</td><td>$f(1) = 1$,</td></tr>
<tr>
<td>     </td><td>(iii)     </td><td>$f(x) + f(y) \le f(x + y)$ whenever $x$, $y$, and $x + y$ are all in $[0, 1]$.</td></tr></tbody></table>Find, with proof, the smallest constant $c$ such that


 $f(x) \le cx$

 for every function $f$ satisfying (i)-(iii) and every $x$ in $[0, 1]$.


 

	\item (\textit{USAMO 1993}) Let $a$, $b$ be odd positive integers. Define the sequence $(f_n)$ by putting $f_1 = a$,
$f_2 = b$, and by letting $f_n$ for $n\ge3$ be the greatest odd divisor of $f_{n-1} + f_{n-2}$.
Show that $f_n$ is constant for $n$ sufficiently large and determine the eventual
value as a function of $a$ and $b$.


 

	\item (\textit{USAMO 1993}) Let $a_0, a_1, a_2,\cdots$ be a sequence of positive real numbers satisfying $a_{i-1}a_{i+1}\le a^2_i$
for $i = 1, 2, 3,\cdots$ . (Such a sequence is said to be \textit{log concave}.) Show that for
each $n > 1$,


 $\frac{a_0+\cdots+a_n}{n+1}\cdot\frac{a_1+\cdots+a_{n-1}}{n-1}\ge\frac{a_0+\cdots+a_{n-1}}{n}\cdot\frac{a_1+\cdots+a_{n}}{n}$.

 

\end{enumerate}
