        	\item (\textit{AMC 12 12 2001}) The sum of two numbers is $S$. Suppose $3$ is added to each number and then
each of the resulting numbers is doubled. What is the sum of the final two
numbers?


 $\text{(A)}\ 2S + 3\qquad \text{(B)}\ 3S + 2\qquad \text{(C)}\ 3S + 6 \qquad\text{(D)} 2S + 6 \qquad \text{(E)}\ 2S + 12$


	\item (\textit{AMC 12 12 2001}) Let $P(n)$ and $S(n)$ denote the product and the sum, respectively, of the digits
of the integer $n$. For example, $P(23) = 6$ and $S(23) = 5$. Suppose $N$ is a
two-digit number such that $N = P(N)+S(N)$. What is the units digit of $N$?


 $\text{(A)}\ 2\qquad \text{(B)}\ 3\qquad \text{(C)}\ 6\qquad \text{(D)}\ 8\qquad \text{(E)}\ 9$


	\item (\textit{AMC 12 12 2001}) The state income tax where Kristin lives is levied at the rate of $p\%$ of the first
$\textdollar 28000$ of annual income plus $(p + 2)\%$ of any amount above $\textdollar 28000$. Kristin
noticed that the state income tax she paid amounted to $(p + 0.25)\%$ of her
annual income. What was her annual income?


 $\text{(A)}\ \$28000\qquad \text{(B)}\ \$32000\qquad \text{(C)}\ \$35000\qquad \text{(D)}\ \$42000\qquad \text{(E)}\ \$56000$


	\item (\textit{AMC 12 12 2001}) The mean of three numbers is $10$ more than the least of the numbers and $15$

 $\text{(A)}\ 5\qquad \text{(B)}\ 20\qquad \text{(C)}\ 25\qquad \text{(D)}\ 30\qquad \text{(E)}\ 36$

	\item (\textit{AMC 12 12 2001}) A telephone number has the form $\text{ABC-DEF-GHIJ}$, where each letter represents

$D$, $E$, and $F$ are consecutive even digits; $G$, $H$, $I$, and $J$ are consecutive odd

 $\text{(A)}\ 4\qquad \text{(B)}\ 5\qquad \text{(C)}\ 6\qquad \text{(D)}\ 7\qquad \text{(E)}\ 8$

	\item (\textit{AMC 12 12 2001}) A box contains exactly five chips, three red and two white. Chips are randomly removed one at a time without replacement until all the red chips are drawn or all the white chips are drawn. What is the probability that the last chip drawn is white?

	\item (\textit{AMC 12 12 2001}) The parabola with equation $y=ax^2+bx+c$ and vertex $(h,k)$ is reflected about the line $y=k$. This results in the parabola with equation $y=dx^2+ex+f$. Which of the following equals $a+b+c+d+e+f$?


 $\text{(A) }2b \qquad \text{(B) }2c \qquad \text{(C) }2a+2b \qquad \text{(D) }2h \qquad \text{(E) }2k$


	\item (\textit{AMC 12 12 2001}) Given the nine-sided regular polygon $A_1 A_2 A_3 A_4 A_5 A_6 A_7 A_8 A_9$, how many distinct equilateral triangles in the plane of the polygon have at least two vertices in the set $\{A_1,A_2,\dots,A_9\}$?

	\item (\textit{AMC 12 12 2001}) An insect lives on the surface of a regular tetrahedron with edges of length 1.  $\text{(A) }\frac {1}{2} \sqrt {3} \qquad \text{(B) }1 \qquad \text{(C) }\sqrt {2} \qquad \text{(D) }\frac {3}{2} \qquad \text{(E) }2$
