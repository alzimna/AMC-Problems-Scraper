	\item (\textit{AMC 12 12B 2009}) In a certain year the price of gasoline rose by $20\%$ during January, fell by $20\%$ during February, rose by $25\%$ during March, and fell by $x\%$ during April.  $\textbf{(A)}\ 12\qquad \textbf{(B)}\ 17\qquad \textbf{(C)}\ 20\qquad \textbf{(D)}\ 25\qquad \textbf{(E)}\ 35$


	\item (\textit{AMC 12 12B 2009}) When a bucket is two-thirds full of water, the bucket and water weigh $a$ kilograms. When the bucket is one-half full of water the total weight is $b$ kilograms. In terms of $a$ and $b$, what is the total weight in kilograms when the bucket is full of water?


 $\textbf{(A)}\ \frac23a + \frac13b\qquad \textbf{(B)}\ \frac32a - \frac12b\qquad \textbf{(C)}\ \frac32a + b \qquad \textbf{(D)}\ \frac32a + 2b\qquad \textbf{(E)}\ 3a - 2b$


	\item (\textit{AMC 12 12B 2009}) Triangle $ABC$ has vertices $A = (3,0)$, $B = (0,3)$, and $C$, where $C$ is on the line $x + y = 7$. What is the area of $\triangle ABC$?


 $\textbf{(A)}\ 6\qquad \textbf{(B)}\ 8\qquad \textbf{(C)}\ 10\qquad \textbf{(D)}\ 12\qquad \textbf{(E)}\ 14$


	\item (\textit{AMC 12 12B 2009}) A particular $12$-hour digital clock displays the hour and minute of a day. Unfortunately, whenever it is supposed to display a $1$, it mistakenly displays a $9$.  $\textbf{(A)}\ \frac12\qquad \textbf{(B)}\ \frac58\qquad \textbf{(C)}\ \frac34\qquad \textbf{(D)}\ \frac56\qquad \textbf{(E)}\ \frac {9}{10}$

On which day, just after Millie has placed the seeds, will the birds find that more than half the seeds in the feeder are millet?


 $\textbf{(A)}\ \text{Tuesday}\qquad \textbf{(B)}\ \text{Wednesday}\qquad \textbf{(C)}\ \text{Thursday} \qquad \textbf{(D)}\ \text{Friday}\qquad \textbf{(E)}\ \text{Saturday}$

	\item (\textit{AMC 12 12B 2009}) Trapezoid $ABCD$ has $AD||BC$, $BD = 1$, $\angle DBA = 23^{\circ}$, and $\angle BDC = 46^{\circ}$.  The ratio $BC: AD$ is $9: 5$.  What is $CD$?

	\item (\textit{AMC 12 12B 2009}) Each face of a cube is given a single narrow stripe painted from the center of one edge to the center of its opposite edge.  $\textbf{(A)}\ \frac {1}{8}\qquad \textbf{(B)}\ \frac {3}{16}\qquad \textbf{(C)}\ \frac {1}{4} \qquad \textbf{(D)}\ \frac {3}{8}\qquad \textbf{(E)}\ \frac {1}{2}$


	\item (\textit{AMC 12 12B 2009}) Rachel and Robert run on a circular track. Rachel runs counterclockwise and completes a lap every $90$ seconds, and Robert runs clockwise and completes a lap every $80$ seconds. Both start from the start line at the same time. At some random time between $10$ minutes and $11$ minutes after they begin to run, a photographer standing inside the track takes a picture that shows one-fourth of the track, centered on the starting line. What is the probability that both Rachel and Robert are in the picture?


 $\textbf{(A)}\ \frac {1}{16}\qquad \textbf{(B)}\ \frac18\qquad \textbf{(C)}\ \frac {3}{16} \qquad \textbf{(D)}\ \frac14\qquad \textbf{(E)}\ \frac {5}{16}$

	\item (\textit{AMC 12 12B 2009}) A convex polyhedron $Q$ has vertices $V_1,V_2,\ldots,V_n$, and $100$ edges.  $\textbf{(A)}\ 200\qquad \textbf{(B)}\ 2n\qquad \textbf{(C)}\ 300\qquad \textbf{(D)}\ 400\qquad \textbf{(E)}\ 4n$
